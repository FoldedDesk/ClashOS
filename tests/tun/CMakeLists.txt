cmake_minimum_required(VERSION 3.16)
project(clahsos_tun_tests C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(CORE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../entry/src/main/cpp" ABSOLUTE)
set(YAML_ROOT ${CORE_ROOT}/third_party/libyaml)
file(GLOB YAML_SOURCES ${YAML_ROOT}/*.c)
add_library(clahsos_yaml STATIC ${YAML_SOURCES})
target_include_directories(clahsos_yaml PUBLIC ${YAML_ROOT})
target_compile_definitions(clahsos_yaml PRIVATE YAML_VERSION_MAJOR=0 YAML_VERSION_MINOR=2
    YAML_VERSION_PATCH=5 YAML_VERSION_STRING="0.2.5" YAML_DECLARE_STATIC)
set(CLAHSOS_TUN_HOST_TEST ON)
add_subdirectory(${CORE_ROOT}/tun engine)
add_subdirectory(${CORE_ROOT}/crypto crypto)
add_subdirectory(${CORE_ROOT}/third_party/zstd zstd)
add_subdirectory(${CORE_ROOT}/third_party/libmaxminddb maxminddb)
add_subdirectory(${CORE_ROOT}/regex regex)
add_subdirectory(${CORE_ROOT}/script script)
add_executable(tun_tests tun_test.cpp ${CORE_ROOT}/tun/clahsos_tun.cpp)
target_include_directories(tun_tests PRIVATE ${CORE_ROOT}/tun)
target_link_libraries(tun_tests PRIVATE clahsos_tunnel_engine)
target_link_libraries(tun_tests PRIVATE clahsos_crypto clahsos_zstd clahsos_maxminddb re2::re2 clashos_quickjs)

find_package(ZLIB REQUIRED)
target_link_libraries(tun_tests PRIVATE ZLIB::ZLIB)
