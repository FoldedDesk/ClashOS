include(GNUInstallDirs)
get_filename_component(MBEDTLS_DIR "${CMAKE_CURRENT_LIST_DIR}/../third_party/mbedtls" ABSOLUTE)
set(MBEDTLS_TARGET_PREFIX clahsos_)
set(MBEDTLS_USER_CONFIG_FILE "${CMAKE_CURRENT_LIST_DIR}/mbedtls_user_config.h")
set(GEN_FILES OFF)
set(USE_STATIC_MBEDTLS_LIBRARY ON)
set(USE_SHARED_MBEDTLS_LIBRARY OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(${MBEDTLS_DIR}/3rdparty mbed-third-party)
add_subdirectory(${MBEDTLS_DIR}/library mbed-library)
add_library(clahsos_sha3 STATIC ../third_party/tiny_sha3/sha3.c)
target_include_directories(clahsos_sha3 PUBLIC ../third_party/tiny_sha3)
add_library(clahsos_crypto INTERFACE)
target_link_libraries(clahsos_crypto INTERFACE clahsos_mbedtls clahsos_sha3)
